		custom_tooltip = {
			text = je_turkfm_tanzimat_main_on_complete_ottomanist

			}
		}
		custom_tooltip = {
			text = je_turkfm_tanzimat_main_on_complete_islamist

			}
		}
		turkfm_cleanup_tanzimat_identity_effect = yes
	}

		OR = {
			custom_tooltip = {
				text = je_turkfm_tanzimat_main_fail
				var:turkfm_tanzimat_reform_var <= 0
			}
			trigger_if = {
				limit = {
					has_variable = turkfm_young_turks_started
				}
				custom_tooltip = {
					text = turkfm_young_turks_failed
					has_variable = turkfm_young_turks_failed
				}
			}
		}
	}

	on_fail = {
		set_global_variable = turkfm_tanzimat_failed
		if = {
			limit = {
				has_variable = turkfm_young_turks_started
			}
			trigger_event = turkfm_tanzimat.007
		}
		else = {
			trigger_event = turkfm_tanzimat.006
		}
		custom_tooltip = turkfm_grand_collapse_possible
		turkfm_cleanup_tanzimat_identity_effect = yes
	}

	on_weekly_pulse = {
		effect = {
			turkfm_calculate_tanzimat_identity_variables_effect = yes

				name = turkfm_tanzimat_identity_var

					name = turkfm_imperial_identity_progress_bar

			}
		}
	}

		effect = {

				name = turkfm_tanzimat_reform_var

				min = turkfm_tanzimat_reformism_min_value
			}
			je:je_turkfm_tanzimat_main = {

					name = turkfm_tanzimat_reforms_progress_bar
					value = owner.var:turkfm_tanzimat_reform_var
				}
			}
		}
	}
}
