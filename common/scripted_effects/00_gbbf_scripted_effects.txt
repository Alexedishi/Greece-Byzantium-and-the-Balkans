####---------- Variable Management Effects ----------####
# NOTE: These effects are meant to execute within the *journal_entry* scope
# It is policy for GotB to set all journal-related variables to be within the journal scope

gbbf_calc_variable_effect = {
	set_variable = {
		name = $name$_var
		value = owner.$name$_value
	}
}

gbbf_push_variable_effect = {
# For use when the calc_ and push_ effects are split between different pulses
	set_variable = {
		name = $name$_var
		value = {
			if = {
				limit = {
					has_variable = $name$_var
				}
				add = var:$name$_var
			}
			add = var:$name$_progress_var
			max = $max$
			min = $min$
		}
	}
}

gbbf_calc_and_push_variable_effect = {
# This effect will restore variables if lost
	set_variable = {
		name = $name$_progress_var
		value = owner.$name$_progress_value
	}
	set_variable = {
		name = $name$_var
		value = {
			if = {
				limit = {
					has_variable = $name$_var
				}
				add = var:$name$_var
			}
			add = var:$name$_progress_var
			max = $max$
			min = $min$
		}
	}
}

gbbf_checked_calc_and_push_variable_effect = {
# This is more nuanced, and will set the base _var to 0 if lost
	if = {
		limit = {
			has_variable = $name$_var
		}
		set_variable = {
			name = $name$_progress_var
			value = owner.$name$_progress_value
		}
		set_variable = {
			name = $name$_var
			value = {
				if = {
					limit = {
						has_variable = $name$_var
					}
					add = var:$name$_var
				}
				add = var:$name$_progress_var
				max = $max$
				min = $min$
			}
		}
	}
	else = {
		set_variable = {
			name = $name$_var
			value = 0
		}
		set_variable = {
			name = $name$_progress_var
			value = owner.$name$_progress_value
		}
	}
}

gbbf_gated_calc_and_push_variable_effect = {
# This will only fire if the base _var is set; it will not restore it if it is lost somehow
	if = {
		limit = {
			has_variable = $name$_var
		}
		set_variable = {
			name = $name$_progress_var
			value = owner.$name$_progress_value
		}
		set_variable = {
			name = $name$_var
			value = {
				if = {
					limit = {
						has_variable = $name$_var
					}
					add = var:$name$_var
				}
				add = var:$name$_progress_var
				max = $max$
				min = $min$
			}
		}
	}
}

gbbf_add_state_claim_root_effect = {
	if = {
		limit = {
			NOT = {
				has_claim = $state_region$
			}
			$state_region$ = {
				any_scope_state = {
					count = all
					owner = root
				}
			}
		}
		$state_region$ = {
			add_claim = root
		}
	}
}

gbbf_add_state_claim_country_effect = {
	if = {
		limit = {
			NOT = {
				has_claim = $state_region$
			}
			$state_region$ = {
				any_scope_state = {
					count = all
					owner = $country$
				}
			}
		}
		$state_region$ = {
			add_claim = $country$
		}
	}
}

gbbf_add_georegion_claims_effect = {
	every_state_region_in_$georegion$ = {

			NOT = {
				any_scope_state = {
					has_claim_by = ROOT
				}
			}
		}

	}
}
