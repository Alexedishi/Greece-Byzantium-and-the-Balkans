turkfm_tanzimat.002 = {
	type = country_event
	placement = root
	icon = "gfx/interface/icons/event_icons/event_fez.dds"
	duration = 3
	dlc = dlc_grefm

	title = turkfm_tanzimat.002.t
	desc = turkfm_tanzimat.002.d
	flavor = turkfm_tanzimat.002.f

	gui_window = event_window_1char_adventure
	left_icon = scope:turkfm_ruler_scope

	event_image = {
		texture = "gfx/interface/illustrations/event_backgrounds/gbbf_turkish_palace2.dds"
	}
	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	on_opened_soundeffect = "event:/SFX/Events/misc/1Character_Banner"
	
	trigger = {
		# triggered by turkfm_start_tanzimat_button
	}

	immediate = {
		ruler = {
			save_scope_as = turkfm_ruler_scope
		}
		ig:ig_intelligentsia = {
			leader = {
				save_scope_as = turkfm_reshid_scope
			}
		}
	}

	option = {
		name = turkfm_tanzimat.002.a
		default_option = yes
		show_as_tooltip = {
			turkfm_add_tanzimat_agenda_effect = {
				type = agriculture
			}
			turkfm_add_tanzimat_agenda_effect = {
				type = military
			}
			turkfm_add_tanzimat_agenda_effect = {
				type = civics
			}
			turkfm_add_tanzimat_agenda_effect = {
				type = economy
			}
			turkfm_add_tanzimat_agenda_effect = {
				type = education
			}
			turkfm_add_tanzimat_agenda_effect = {
				type = security
			}
			turkfm_add_tanzimat_agenda_effect = {
				type = slavery
			}
			add_modifier = {
				name = modifier_turkfm_reorganizing_spirit
				months  = 360
			}
			ig:ig_intelligentsia = {
				add_ideology = ideology_anti_slavery
			}
			every_interest_group = {
				limit = {
					law_stance = {
						law = law_type:law_state_religion
						value > neutral
					}
					NOR = {
						law_stance = {
							law = law_type:law_landed_voting
							value > neutral
						}
						law_stance = {
							law = law_type:law_wealth_voting
							value > neutral
						}
						law_stance = {
							law = law_type:law_census_voting
							value > neutral
						}
						law_stance = {
							law = law_type:law_universal_suffrage
							value > neutral
						}
					}
				}
				add_modifier = {
					name = modifier_turkfm_influence_questioned
					months = gbbf_long_modifier_time
					is_decaying = yes
				}
			}
			every_political_movement = {
				limit = {
					OR = {
						is_political_movement_type = movement_religious_majority
						is_political_movement_type = movement_modernizer
					}
				}
				add_modifier = {
					name = modifier_turkfm_declared_reforms
					months = gbbf_very_long_modifier_time
				}
			}
		}
	}
}

turkfm_tanzimat.003 = {
	type = country_event
	placement = root
	icon = "gfx/interface/icons/event_icons/event_portrait.dds"
	duration = 3
	dlc = dlc_grefm

	title = turkfm_tanzimat.003.t
	desc = turkfm_tanzimat.003.d
	flavor = turkfm_tanzimat.003.f

	gui_window = event_window_3char_selection
	left_icon = scope:turkfm_tanzimat_character_1
	center_icon = scope:turkfm_tanzimat_character_2
	right_icon = scope:turkfm_tanzimat_character_3

	event_image = {
		texture = "gfx/interface/illustrations/event_backgrounds/gbbf_turkish_palace1.dds"
	}
	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	on_opened_soundeffect = "event:/SFX/Events/misc/1Character_Banner"
	
	trigger = {
		NOT = {
			has_variable = turkfm_tanzimat_men_happened
		}
		has_global_variable = turkfm_tanzimat_started
		var:turkfm_completed_tanzimat_agendas_var >= 2
	}

	immediate = {
		set_variable = turkfm_tanzimat_men_happened
		if = {
			limit = {
				ig:ig_intelligentsia = {
					leader = {
						has_template = turkfm_reshid_mustafa
					}
				}
			}
			ig:ig_intelligentsia = {
				leader = {
					save_scope_as = turkfm_tanzimat_character_1
				}
			}
		}
		else = {
			create_character = {
				template = turkfm_reshid_mustafa
				on_created = {
					add_character_role = politician
				}
			}
			random_scope_character = {
				limit = {
					has_template = turkfm_reshid_mustafa
				}
				save_scope_as = turkfm_tanzimat_character_1
			}
		}
		create_character = {
			template = turkfm_ali_mehmed
			on_created = {
				add_character_role = politician
			}
		}
		random_scope_character = {
			limit = {
				has_template = turkfm_ali_mehmed
			}
			save_scope_as = turkfm_tanzimat_character_2
		}
		create_character = {
			template = turkfm_fuad_mehmed
			on_created = {
				add_character_role = politician
			}
		}
		random_scope_character = {
			limit = {
				has_template = turkfm_fuad_mehmed
			}
			save_scope_as = turkfm_tanzimat_character_3
		}
	}

	option = {
		name = turkfm_tanzimat.003.a
		default_option = yes
		add_modifier = {
			name = modifier_turkfm_reorganizing_spirit
			months = gbbf_very_long_modifier_time
		}
		if = {
			limit = {
				ig:ig_intelligentsia = {
					leader = scope:turkfm_tanzimat_character_1
				}
			}
			scope:turkfm_tanzimat_character_1 = {
				add_trait = gbbf_trait_master_bureaucrat
			}
		}
		else = {
			scope:turkfm_tanzimat_character_1 = {
				add_modifier = {
					name = modifier_turkfm_reformist_leader
					months = gbbf_very_long_modifier_time
					is_decaying = yes
				}
			}
		}
		scope:turkfm_tanzimat_character_2 = {
			add_modifier = {
				name = modifier_turkfm_reformist_leader
				months = gbbf_very_long_modifier_time
				is_decaying = yes
			}
		}
		scope:turkfm_tanzimat_character_3 = {

				name = modifier_turkfm_reformist_leader

turkfm_tanzimat.010 = {
	type = country_event
	placement = root
	icon = "gfx/interface/icons/event_icons/event_industry.dds"
	duration = 3
	dlc = dlc_grefm

	title = turkfm_tanzimat.010.t
	desc = turkfm_tanzimat.010.d
	flavor = turkfm_tanzimat.010.f

	event_image = {
		video = "southamerica_factory_opening"
	}
	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	
	trigger = {
		has_variable = turkfm_tanzimat_agriculture_agenda
		NOR = {
			has_law_or_variant = law_type:law_traditionalism
			has_law_or_variant = law_type:law_agrarianism
		}
		active_law:lawgroup_land_reform ?= {
			NOT = {
				has_amendment = amendment_type:amendment_gbbf_chiftlik
			}
		}
		any_scope_building = {
			is_building_type = building_wheat_farm
			NOT = {
				has_active_production_method = pm_simple_farming
			}
			level >= 10
			weekly_profit > 5
			occupancy >= 0.9
		}
	}
	
	cooldown = {
		years = 1
	}

	immediate = {
	}

	option = {
		name = turkfm_tanzimat.010.a
		default_option = yes
		turkfm_complete_tanzimat_agenda_effect = {
			type = agriculture
		}
		if = {
			limit = {
				has_modifier = modifier_turkfm_feudal_argiculture
			}
			remove_modifier = modifier_turkfm_feudal_argiculture
		}
		every_political_movement = {
			limit = {
				OR = {
					law_stance = {
						law = law_type:law_traditionalism
						value < neutral
					}
					law_stance = {
						law = law_type:law_agrarianism
						value < neutral
					}
				}
			}

	type = country_event

	dlc = dlc_grefm

	title = turkfm_tanzimat.200.t
	desc = turkfm_tanzimat.200.d
	flavor = turkfm_tanzimat.200.f

	event_image = {
		video = "africa_leader_arguing"
	}
	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	trigger = {
		# triggered by je_turkfm_tanzimat_young_turks
	}

	immediate = {
		if = {
			limit = {
				NOT = {
					any_political_movement = {
						is_political_movement_type = movement_gbbf_young_turks
					}
				}
			}
			create_political_movement = { type = movement_gbbf_young_turks }
		}
		random_political_movement = {
			limit = {
				is_political_movement_type = movement_gbbf_young_turks
			}
			save_scope_as = young_turks_movement_scope
		}
		random_country = {
			limit = {
				is_in_geographic_region = geographic_region_europe
				country_rank >= rank_value:great_power
				relations:root <= relations_threshold:neutral
				NOT = {
					has_law = law_type:law_outlawed_dissent
				}
			}
			save_scope_as = young_turks_host_scope
		}
		if = {
			limit = {
				NOT = {
					exists = scope:young_turks_host_scope
				}
			}
			random_country = {
				limit = {
					is_in_geographic_region = geographic_region_europe
					country_rank >= rank_value:minor_power
					NOT = {
						has_law = law_type:law_outlawed_dissent
					}
				}
				save_scope_as = young_turks_host_scope
			}
		}
		create_character = {
			template = turkfm_sabahaddin_mehmed
			on_created = {
				save_scope_as = turkfm_sabahaddin_scope
				if = {
					limit = {
						# PDX implemented this character before GotB. He is for flavor-only if VOTP is not present
						has_dlc_feature = voice_of_the_people_content
					}
					scope:turkfm_sabahaddin_scope = {
						add_character_role = agitator
					}
				}
				else = {
					place_character_in_void = 240
				}
			}
		}
		create_character = {
			template = turkfm_sojliu_ibrahim
			on_created = {
				save_scope_as = young_turks_leader_scope
				add_character_role = politician
				add_character_role = agitator
			}
		}
	}

	option = {
		name = turkfm_tanzimat.200.a
		default_option = yes
		scope:young_turks_movement_scope = {
			set_core_ideology = ideology_positivist_movement
		}
		show_as_tooltip = {
			add_modifier = modifier_turkfm_political_instability
		}
	}
}
